        if selectedButton then

            selectedButton.BackgroundColor3 = selectedColor

        end

        

        -- Select new button

        selectedButton = button

        selectedPlayer = player

        button.BackgroundColor3 = Color3.fromRGB(60, 60, 80)

        

        -- Hide no player frame and show selected player frame

        NoPlayerFrame.Visible = false

        SelectedPlayerFrame.Visible = true

        

        -- Update selected player info

        SelectedAvatar.Image = "rbxthumb://type=AvatarHeadShot&id=" .. player.UserId .. "&w=150&h=150"

        SelectedName.Text = player.Name

        

        -- Show action buttons

        TeleportButton.Visible = true

        FollowButton.Visible = not isFollowing

        StopFollowButton.Visible = isFollowing

    end)

    

    -- Hover effect

    button.MouseEnter:Connect(function()

        if selectedButton ~= button then

    end)

local function teleportToPlayer()

    if selectedPlayer and selectedPlayer.Character and selectedPlayer.Character:FindFirstChild("HumanoidRootPart") then

        local targetPosition = selectedPlayer.Character.HumanoidRootPart.Position

        if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then

            LocalPlayer.Character.HumanoidRootPart.CFrame = CFrame.new(targetPosition + Vector3.new(0, 0, 3))

        end

    end

end

local function startFollowing()

    if selectedPlayer and selectedPlayer.Character then

        isFollowing = true

        FollowButton.Visible = false

        StopFollowButton.Visible = true

        

        followConnection = RunService.Heartbeat:Connect(function()

            if selectedPlayer and selectedPlayer.Character and selectedPlayer.Character:FindFirstChild("HumanoidRootPart") then

                local targetPosition = selectedPlayer.Character.HumanoidRootPart.Position

                if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then

                    LocalPlayer.Character.HumanoidRootPart.CFrame = CFrame.new(targetPosition + Vector3.new(0, 5, 3))

                end

            else

                stopFollowing()

            end

        end)

    end

end

local function stopFollowing()

    if followConnection then

        followConnection:Disconnect()

        followConnection = nil

    end

    isFollowing = false

    FollowButton.Visible = true

    StopFollowButton.Visible = false

end

local function cancelSelection()

    if selectedButton then

        selectedButton.BackgroundColor3 = selectedColor

        selectedButton = nil

    end

    selectedPlayer = nil

    stopFollowing()

    NoPlayerFrame.Visible = true

    SelectedPlayerFrame.Visible = false

    TeleportButton.Visible = false

    FollowButton.Visible = false

    StopFollowButton.Visible = false

end

local function updatePlayerList()

    -- Clear existing buttons

    for _, child in pairs(PlayerList:GetChildren()) do

        if child:IsA("TextButton") then

            child:Destroy()

        end

    end

    

    -- Create buttons for all players

    for _, player in pairs(Players:GetPlayers()) do

        if player ~= LocalPlayer then

            createPlayerButton(player)

        end

    end

    

    -- Update canvas size

    PlayerList.CanvasSize = UDim2.new(0, 0, 0, UIListLayout.AbsoluteContentSize.Y)

end
