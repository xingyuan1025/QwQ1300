-- Toggle logic
local minimized = false
toggle.MouseButton1Click:Connect(function()
	minimized = not minimized
	scroll.Visible = not minimized
	refresh.Visible = not minimized
	toggle.Text = minimized and "+" or "-"
	frame.Size = minimized and UDim2.new(0, 520, 0, 60) or UDim2.new(0, 520, 0, 400)
end)

-- Copy helper
local function copy(text)
	if setclipboard then
		setclipboard(text)
	end
end

-- Clear previous
local function clear()
	for _, v in ipairs(scroll:GetChildren()) do
		if v:IsA("Frame") then v:Destroy() end
	end
end

-- Update Sound List
local function update()
	clear()
	local allSounds = {}
	for _, obj in ipairs(game:GetDescendants()) do
		if obj:IsA("Sound") and obj.SoundId ~= "" then
			table.insert(allSounds, obj)
		end
	end

	for _, sound in ipairs(allSounds) do
		local row = Instance.new("Frame", scroll)
		row.Size = UDim2.new(1, 0, 0, 50)
		row.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
		row.BorderColor3 = Color3.fromRGB(0, 0, 0)

		local label = Instance.new("TextLabel", row)
		label.Size = UDim2.new(0.7, -10, 1, 0)
		label.Position = UDim2.new(0, 5, 0, 0)
		label.BackgroundTransparency = 1
		label.Text = sound.Name .. " | " .. sound.SoundId
		label.TextXAlignment = Enum.TextXAlignment.Left
		label.TextScaled = true
		label.Font = Enum.Font.SourceSans
		label.TextColor3 = Color3.fromRGB(0, 0, 0)

		local copyBtn = Instance.new("TextButton", row)
		copyBtn.Text = "复制"
		copyBtn.Size = UDim2.new(0.3, -10, 0.8, 0)
		copyBtn.Position = UDim2.new(0.7, 5, 0.1, 0)
		copyBtn.BackgroundColor3 = Color3.fromRGB(200, 255, 200)
		copyBtn.BorderColor3 = Color3.fromRGB(0, 0, 0)
		copyBtn.TextScaled = true
		copyBtn.Font = Enum.Font.SourceSansBold

		copyBtn.MouseButton1Click:Connect(function()
			copy(sound.SoundId)
		end)
	end
end

-- Connect refresh
refresh.MouseButton1Click:Connect(update)

-- Initial update
update()
